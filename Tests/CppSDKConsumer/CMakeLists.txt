cmake_minimum_required(VERSION 3.24)
set(CMAKE_CONFIGURATION_TYPES "RelWithDebInfo;Release" CACHE STRING "SDK-compatible configurations")
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL CACHE STRING "SDK DLL CRT")
project(WorkphoneCppSDKConsumer LANGUAGES CXX)
find_package(WorkphoneCppSDK 0.1 CONFIG REQUIRED)
add_executable(cpp-sdk-consumer main.cpp)
target_link_libraries(cpp-sdk-consumer PRIVATE WorkphoneCpp::Graphics)
add_executable(cpp-sdk-cooked-consumer cooked_content.cpp)
target_link_libraries(cpp-sdk-cooked-consumer PRIVATE WorkphoneCpp::Graphics)
set(CPP_SDK_CONTENT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../content" CACHE PATH "Cooked runtime package")
option(CPP_SDK_BUILD_COOKER "Build the catalog-backed sample authoring tool" OFF)
if(CPP_SDK_BUILD_COOKER)
    add_executable(cpp-sdk-cook-sample cook_sample.cpp)
    target_link_libraries(cpp-sdk-cook-sample PRIVATE WorkphoneCpp::Graphics WorkphoneCpp::SQLite)
endif()
enable_testing()
add_test(NAME cpp-sdk-consumer COMMAND cpp-sdk-consumer)
add_test(NAME cpp-sdk-cooked-consumer COMMAND cpp-sdk-cooked-consumer "${CPP_SDK_CONTENT_DIRECTORY}")
# Resolve DLL imports from the extracted package rather than the checkout.
get_target_property(_sdk_dll WorkphoneCpp::Core IMPORTED_LOCATION_RELWITHDEBINFO)
get_filename_component(_sdk_bin "${_sdk_dll}" DIRECTORY)
set_tests_properties(cpp-sdk-consumer cpp-sdk-cooked-consumer PROPERTIES
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_sdk_bin}" TIMEOUT 60)
