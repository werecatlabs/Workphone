set(WP_NETWORK_TEST_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(WorkphoneNetworkTests NativeNetworkTests.c)
target_link_libraries(WorkphoneNetworkTests PRIVATE WorkphoneNetwork)
set_target_properties(WorkphoneNetworkTests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON FOLDER Tests/Network)
add_test(NAME WorkphoneNetwork.native COMMAND WorkphoneNetworkTests)
set_tests_properties(WorkphoneNetwork.native PROPERTIES LABELS "network;headless" TIMEOUT 30)
add_executable(WPNetworkCodecTests NetworkCodecTests.cpp)
target_include_directories(WPNetworkCodecTests PRIVATE "${WP_NETWORK_TEST_ROOT}/Engine/cpp/Include")
target_compile_features(WPNetworkCodecTests PRIVATE cxx_std_17)
set_target_properties(WPNetworkCodecTests PROPERTIES FOLDER Tests/Network)
add_test(NAME WPNetwork.codec COMMAND WPNetworkCodecTests)
set_tests_properties(WPNetwork.codec PROPERTIES LABELS "network;headless" TIMEOUT 30)
if(MSVC)
    target_compile_options(WorkphoneNetworkTests PRIVATE /W4 /WX)
    target_compile_options(WPNetworkCodecTests PRIVATE /W4 /WX)
else()
    target_compile_options(WorkphoneNetwork PRIVATE -Wall -Wextra -Werror)
    target_compile_options(WorkphoneNetworkTests PRIVATE -Wall -Wextra -Werror)
    target_compile_options(WPNetworkCodecTests PRIVATE -Wall -Wextra -Werror)
endif()
