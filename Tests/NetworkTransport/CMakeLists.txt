cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
project(WPNetworkTransportConformance LANGUAGES CXX)
enable_testing()
set(Protobuf_USE_STATIC_LIBS ON)
set(CMAKE_MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
set(CMAKE_MAP_IMPORTED_CONFIG_MINSIZEREL Release)
find_package(Protobuf CONFIG REQUIRED)
find_package(GameNetworkingSockets CONFIG REQUIRED)
add_executable(WPNetworkGNSTests GNSConformance.cpp)
target_compile_features(WPNetworkGNSTests PRIVATE cxx_std_17)
target_link_libraries(WPNetworkGNSTests PRIVATE GameNetworkingSockets::static)
if(MSVC)
    target_compile_options(WPNetworkGNSTests PRIVATE /W4 /WX /EHsc)
endif()
add_test(NAME WPNetwork.gns_transport COMMAND WPNetworkGNSTests)
set_tests_properties(WPNetwork.gns_transport PROPERTIES TIMEOUT 30 LABELS "network;transport;experimental")

# Standalone operator tool. The signing key is never linked or packaged with a game.
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(WPNetworkGNSIssuer GNSIssuer.cpp GNSIssuerWire.proto)
protobuf_generate(TARGET WPNetworkGNSIssuer)
target_compile_features(WPNetworkGNSIssuer PRIVATE cxx_std_17)
target_include_directories(WPNetworkGNSIssuer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(WPNetworkGNSIssuer PRIVATE protobuf::libprotobuf OpenSSL::Crypto)
