cmake_minimum_required(VERSION 3.21)
project(WPAudioContracts LANGUAGES C CXX)
include(CTest)
if(NOT TARGET WorkphoneAudio)
    add_subdirectory(../../Engine/c/Project/WorkphoneAudio native_audio)
endif()
add_executable(WPAudioCoreTests AudioCoreTests.cpp)
target_compile_features(WPAudioCoreTests PRIVATE cxx_std_17)
target_link_libraries(WPAudioCoreTests PRIVATE WorkphoneAudio)
foreach(suite contracts decode render)
    add_test(NAME WPAudio.${suite} COMMAND WPAudioCoreTests ${suite})
    set_tests_properties(WPAudio.${suite} PROPERTIES LABELS "audio;offline;mandatory" TIMEOUT 30)
endforeach()
if(TARGET Workphone)
    add_executable(WPAudioSceneTests AudioSceneTests.cpp)
    target_compile_features(WPAudioSceneTests PRIVATE cxx_std_17)
    target_compile_definitions(WPAudioSceneTests PRIVATE BOOST_ALL_NO_LIB)
    target_include_directories(WPAudioSceneTests PRIVATE
        ${WP_ROOT}/Engine/cpp/Include ${WP_ROOT}/Dependencies/boost)
    target_link_libraries(WPAudioSceneTests PRIVATE Workphone)
    target_link_directories(WPAudioSceneTests PRIVATE
        ${WP_ROOT}/libs/windows/${WP_TOOLSET}/${WP_ARCH}/${WP_STATIC_LINK_CRT_TYPE}/$<CONFIG>)
    add_test(NAME WPAudio.scene COMMAND WPAudioSceneTests)
    set_tests_properties(WPAudio.scene PROPERTIES LABELS "audio;scene;offline" TIMEOUT 30)
endif()
if(TARGET WPAudio)
    add_executable(WPAudioAdapterTests AudioAdapterTests.cpp)
    target_compile_features(WPAudioAdapterTests PRIVATE cxx_std_17)
    target_compile_definitions(WPAudioAdapterTests PRIVATE BOOST_ALL_NO_LIB)
    target_include_directories(WPAudioAdapterTests PRIVATE
        ${WP_ROOT}/Engine/cpp/Include ${WP_ROOT}/Dependencies/boost)
    target_link_libraries(WPAudioAdapterTests PRIVATE WPAudio Workphone)
    target_link_directories(WPAudioAdapterTests PRIVATE
        ${WP_ROOT}/libs/windows/${WP_TOOLSET}/${WP_ARCH}/${WP_STATIC_LINK_CRT_TYPE}/$<CONFIG>)
    add_test(NAME WPAudio.adapter COMMAND WPAudioAdapterTests)
    set_tests_properties(WPAudio.adapter PROPERTIES LABELS "audio;adapter;offline" TIMEOUT 30)
endif()
if(WIN32)
    add_executable(WPAudioXAudio2Spike XAudio2CoreSpike.cpp)
    target_compile_features(WPAudioXAudio2Spike PRIVATE cxx_std_17)
    target_link_libraries(WPAudioXAudio2Spike PRIVATE WorkphoneAudio xaudio2 ole32)
    add_test(NAME WPAudio.device COMMAND WPAudioXAudio2Spike)
    set_tests_properties(WPAudio.device PROPERTIES LABELS "audio;hardware;spike"
        SKIP_RETURN_CODE 77 TIMEOUT 15)
endif()
if(MSVC)
    target_compile_options(WorkphoneAudio PRIVATE /W4 /WX)
    target_compile_options(WPAudioCoreTests PRIVATE /W4 /WX)
else()
    target_compile_options(WorkphoneAudio PRIVATE -Wall -Wextra -Werror)
    target_compile_options(WPAudioCoreTests PRIVATE -Wall -Wextra -Werror)
endif()
