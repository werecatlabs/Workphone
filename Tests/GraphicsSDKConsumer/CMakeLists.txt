cmake_minimum_required(VERSION 3.24)
project(GraphicsSDKConsumer LANGUAGES C CXX)
enable_testing()
if(NOT SDK_ROOT)
    message(FATAL_ERROR "Pass SDK_ROOT pointing to the relocated installation")
endif()
find_package(WorkphoneGraphicsSDK 0.1 CONFIG REQUIRED
    PATHS "${SDK_ROOT}/lib/cmake/WorkphoneGraphicsSDK" NO_DEFAULT_PATH)
# Every exported header/archive path must resolve within the moved package.
file(REAL_PATH "${SDK_ROOT}" _root)
foreach(_target Core Graphics PlatformWin32)
    get_target_property(_includes WorkphoneGraphics::${_target} INTERFACE_INCLUDE_DIRECTORIES)
    get_target_property(_configs WorkphoneGraphics::${_target} IMPORTED_CONFIGURATIONS)
    set(_paths ${_includes})
    foreach(_config IN LISTS _configs)
        get_target_property(_archive WorkphoneGraphics::${_target} IMPORTED_LOCATION_${_config})
        list(APPEND _paths "${_archive}")
    endforeach()
    foreach(_path IN LISTS _paths)
        file(REAL_PATH "${_path}" _resolved)
        cmake_path(IS_PREFIX _root "${_resolved}" NORMALIZE _inside)
        if(NOT _inside OR NOT EXISTS "${_resolved}")
            message(FATAL_ERROR "Export escapes installation or is missing: ${_path}")
        endif()
    endforeach()
endforeach()
add_executable(graphics_sdk_c90 consumer.c)
set_target_properties(graphics_sdk_c90 PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
add_executable(graphics_sdk_cpp17 consumer.cpp)
set_target_properties(graphics_sdk_cpp17 PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
foreach(_target graphics_sdk_c90 graphics_sdk_cpp17)
    target_link_libraries(${_target} PRIVATE WorkphoneGraphics::SDK)
    set_target_properties(${_target} PROPERTIES
        MSVC_RUNTIME_LIBRARY "${WorkphoneGraphicsSDK_MSVC_RUNTIME_LIBRARY}")
    add_test(NAME ${_target} COMMAND ${_target})
endforeach()
